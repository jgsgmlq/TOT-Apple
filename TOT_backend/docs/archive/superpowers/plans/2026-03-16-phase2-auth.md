```javascript
// server/auth.js
// 职责：JWT 签发、验证中间件、白名单配置
// 用法：在 index.js 中 app.use(authMiddleware) 注册；白名单路由跳过验证

const jwt = require('jsonwebtoken');

const JWT_SECRET = process.env.JWT_SECRET || 'mvp-dev-secret-change-in-prod';
const JWT_EXPIRES = '7d';

if (!process.env.JWT_SECRET) {
  console.warn('[auth] WARNING: JWT_SECRET not set, using default dev secret. DO NOT use in production.');
}

// 白名单：这些路由无需 token
const PUBLIC_PATHS = [
  { method: 'GET',  path: '/api/health' },
  { method: 'POST', path: '/api/login' },
  { method: 'GET',  path: '/api/state' },
  { method: 'GET',  path: '/api/events' },
  { method: 'POST', path: '/api/events' },
];

function isPublic(req) {
  return PUBLIC_PATHS.some(p =>
    p.method === req.method && req.path === p.path
  );
}

// JWT 中间件
function authMiddleware(req, res, next) {
  // 静态文件和 WebSocket 升级不经过此中间件
  if (isPublic(req)) return next();

  const authHeader = req.headers['authorization'];
  if (!authHeader || !authHeader.startsWith('Bearer ')) {
    return res.status(401).json({ error: 'Unauthorized' });
  }

  const token = authHeader.slice(7);
  try {
    const decoded = jwt.verify(token, JWT_SECRET);
    req.user = decoded; // { id, role, storeId, employeeId, name }
    next();
  } catch (err) {
    return res.status(401).json({ error: 'Unauthorized' });
  }
}

// 签发 token
function signToken(payload) {
  return jwt.sign(payload, JWT_SECRET, { expiresIn: JWT_EXPIRES, algorithm: 'HS256' });
}

// 角色守卫中间件工厂
function requireRole(...roles) {
  return (req, res, next) => {
    if (!req.user || !roles.includes(req.user.role)) {
      return res.status(403).json({ error: 'Forbidden' });
    }
    next();
  };
}

module.exports = { authMiddleware, signToken, requireRole };
```

```javascript
// server/routes/login.js
// 职责：POST /api/login（登录验证+签发 JWT）、GET /api/me（当前用户信息）

const express = require('express');
const bcrypt = require('bcryptjs');
const db = require('../db');
const { signToken } = require('../auth');

const router = express.Router();

// POST /api/login
router.post('/login', (req, res) => {
  const { username, password } = req.body || {};
  if (!username || !password) {
    return res.status(400).json({ error: '请输入用户名和密码' });
  }

  const user = db.prepare(
    'SELECT id, username, password_hash, role, store_id, employee_id FROM users WHERE username = ?'
  ).get(username);

  if (!user) {
    return res.status(401).json({ error: '用户名或密码错误' });
  }

  const valid = bcrypt.compareSync(password, user.password_hash);
  if (!valid) {
    return res.status(401).json({ error: '用户名或密码错误' });
  }

  // 查询显示名称
  let name = username;
  if (user.employee_id) {
    const emp = db.prepare('SELECT name FROM employees WHERE id = ?').get(user.employee_id);
    if (emp) name = emp.name;
  } else if (user.role === 'central') {
    name = '总部运营';
  } else if (user.role === 'admin') {
    name = '系统管理员';
  }

  const payload = {
    id: user.id,
    role: user.role,
    storeId: user.store_id || null,
    employeeId: user.employee_id || null,
    name,
  };
  const token = signToken(payload);

  res.json({ token, user: payload });
});

// GET /api/me（需要 authMiddleware，由 index.js 保障）
router.get('/me', (req, res) => {
  res.json(req.user);
});

module.exports = router;
```

```javascript
// front-end/js/auth.js
// 职责：JWT 存取、auth 检查、登出工具函数
// 依赖：无外部库，纯 Vanilla JS
// 用法：在各 HTML 页面 <head> 中最先引入，然后立即调用 Auth.requireRole('xxx')

(function () {
  const TOKEN_KEY = 'mvp_token';
  const USER_KEY  = 'mvp_user';

  // 从 JWT payload 中解码用户信息（不验证签名，仅用于客户端判断）
  function decodeToken(token) {
    try {
      const payload = token.split('.')[1];
      // Base64Url → Base64 → JSON
      const base64 = payload.replace(/-/g, '+').replace(/_/g, '/');
      const json = decodeURIComponent(
        atob(base64).split('').map(c => '%' + ('00' + c.charCodeAt(0).toString(16)).slice(-2)).join('')
      );
      return JSON.parse(json);
    } catch (e) {
      return null;
    }
  }

  function getToken() { return localStorage.getItem(TOKEN_KEY); }
  function getUser()  {
    try { return JSON.parse(localStorage.getItem(USER_KEY) || 'null'); } catch { return null; }
  }

  function isTokenValid() {
    const token = getToken();
    if (!token) return false;
    const decoded = decodeToken(token);
    if (!decoded || !decoded.exp) return false;
    // 检查是否过期（服务器时间可能不同，留 30s buffer）
    return decoded.exp > Math.floor(Date.now() / 1000) + 30;
  }

  // 保存登录信息
  function saveAuth(token, user) {
    localStorage.setItem(TOKEN_KEY, token);
    localStorage.setItem(USER_KEY, JSON.stringify(user));
  }

  // 清除登录信息
  function clearAuth() {
    localStorage.removeItem(TOKEN_KEY);
    localStorage.removeItem(USER_KEY);
  }

  // 登出并跳转登录页
  function logout() {
    clearAuth();
    window.location.replace('login.html');
  }

  // 角色检查（同步，无网络请求）
  // 调用后立即执行；token 无效或 role 不匹配则立即跳转 login.html
  function requireRole(requiredRole) {
    if (!isTokenValid()) {
      window.location.replace('login.html');
      return;
    }
    const user = getUser();
    if (!user || (requiredRole && user.role !== requiredRole && user.role !== 'admin')) {
      window.location.replace('login.html');
      return;
    }
  }

  // 返回用于 API 调用的 Authorization header
  function authHeaders() {
    const token = getToken();
    return token ? { 'Authorization': 'Bearer ' + token, 'Content-Type': 'application/json' } : {};
  }

  window.Auth = { requireRole, saveAuth, clearAuth, logout, getToken, getUser, authHeaders, isTokenValid };
})();
```
